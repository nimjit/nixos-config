* Typst Math Preview

Custom implementation rendering =$...$= Typst fragments as inline SVG overlays.
Cursor leaves fragment → renders via =typst compile --format svg=.
Cursor enters fragment → shows source for editing.

This mirrors typst-mate behavior in Obsidian, using the Typst CLI already installed.

#+begin_src emacs-lisp
(defgroup org-typst-preview nil
  "Preview Typst math fragments inline in org-mode."
  :group 'org)

(defcustom org-typst-preview-scale 0.9
  "SVG scale factor for rendered math."
  :type 'number :group 'org-typst-preview)

;; Ukiyo base05 (main text color) — update if you change themes
(defcustom org-typst-preview-text-color "#ccc2b7"
  "Typst text color for rendered fragments; should match your theme."
  :type 'string :group 'org-typst-preview)

(defvar-local org-typst-preview--last-point nil)

(defun org-typst-preview--find-fragment (pos)
  "Return (beg . end) of the $...$ fragment containing POS, or nil.
Only matches single-line fragments without nested dollar signs."
  (save-excursion
    (let ((line-beg (line-beginning-position))
          (line-end (line-end-position))
          result)
      (goto-char line-beg)
      (while (and (not result)
                  (re-search-forward "\\$\\([^$\n]+\\)\\$" line-end t))
        (when (and (>= pos (match-beginning 0))
                   (<= pos (match-end 0)))
          (setq result (cons (match-beginning 0) (match-end 0)))))
      result)))

(defun org-typst-preview--render (beg end)
  "Compile the Typst fragment at BEG..END to SVG and display as an overlay."
  ;; Clear any existing overlay at this location
  (dolist (ov (overlays-in beg end))
    (when (overlay-get ov 'org-typst-preview) (delete-overlay ov)))
  (let* ((fragment (buffer-substring-no-properties beg end))
         (tmp-dir  (make-temp-file "org-typst-" t))
         (typ-file (expand-file-name "math.typ" tmp-dir))
         (svg-file (expand-file-name "math.svg" tmp-dir)))
    (with-temp-file typ-file
      (insert "#set page(width: auto, height: auto, margin: (x: 2pt, y: 3pt), fill: none)\n")
      (insert (format "#set text(size: 10.5pt, fill: rgb(\"%s\"))\n"
                      org-typst-preview-text-color))
      (insert fragment "\n"))
    (when (zerop (call-process "typst" nil nil nil
                               "compile" "--format" "svg" typ-file svg-file))
      (let ((ov (make-overlay beg end)))
        (overlay-put ov 'display
                     (create-image svg-file 'svg nil
                                   :scale org-typst-preview-scale))
        (overlay-put ov 'org-typst-preview t)
        (overlay-put ov 'evaporate t)))))

(defun org-typst-preview-buffer ()
  "Render all $...$ fragments in the current buffer."
  (interactive)
  (dolist (ov (overlays-in (point-min) (point-max)))
    (when (overlay-get ov 'org-typst-preview) (delete-overlay ov)))
  (save-excursion
    (goto-char (point-min))
    (while (re-search-forward "\\$\\([^$\n]+\\)\\$" nil t)
      (let ((beg (match-beginning 0)) (end (match-end 0)))
        ;; Don't render the fragment the cursor is currently inside
        (unless (and (>= (point) beg) (<= (point) end))
          (org-typst-preview--render beg end))))))

(defun org-typst-preview--post-command ()
  "Auto-toggle preview when cursor moves in or out of a math fragment."
  (when (and (eq major-mode 'org-mode) (not (minibufferp)))
    (let ((prev org-typst-preview--last-point)
          (curr (point)))
      (setq org-typst-preview--last-point curr)
      (when prev
        (let ((prev-frag (org-typst-preview--find-fragment prev))
              (curr-frag (org-typst-preview--find-fragment curr)))
          ;; Moved OUT of a fragment → render it
          (when (and prev-frag (not (equal prev-frag curr-frag)))
            (unless (seq-find (lambda (o) (overlay-get o 'org-typst-preview))
                              (overlays-in (car prev-frag) (cdr prev-frag)))
              (org-typst-preview--render (car prev-frag) (cdr prev-frag))))
          ;; Moved INTO a fragment → remove overlay to show source
          (when (and curr-frag (not (equal curr-frag prev-frag)))
            (dolist (ov (overlays-in (car curr-frag) (cdr curr-frag)))
              (when (overlay-get ov 'org-typst-preview) (delete-overlay ov)))))))))

(define-minor-mode org-typst-preview-mode
  "Auto-preview Typst math $...$ fragments in org-mode buffers."
  :lighter " TypPrev"
  (if org-typst-preview-mode
      (progn
        (add-hook 'post-command-hook #'org-typst-preview--post-command nil t)
        (org-typst-preview-buffer))
    (remove-hook 'post-command-hook #'org-typst-preview--post-command t)
    (dolist (ov (overlays-in (point-min) (point-max)))
      (when (overlay-get ov 'org-typst-preview) (delete-overlay ov)))))
#+end_src

* Google Calendar Sync

=org-gcal= provides two-way sync with Google Calendar. Credentials are read from
=~/.authinfo.gpg= (never committed to git).

#+begin_src emacs-lisp
(use-package org-gcal
  :config
  ;; Credentials live in ~/.authinfo.gpg:
  ;; machine org-gcal login client-id password YOUR_CLIENT_ID
  ;; machine org-gcal login client-secret password YOUR_CLIENT_SECRET
  (setq org-gcal-client-id
        (auth-source-pick-first-password :host "org-gcal" :user "client-id"))
  (setq org-gcal-client-secret
        (auth-source-pick-first-password :host "org-gcal" :user "client-secret"))
  (setq org-gcal-fetch-file-alist
        '(("tidemanus@gmail.com" . "~/org/gcal.org")))
  ;; Auto-sync when opening agenda
  (add-hook 'org-agenda-mode-hook #'org-gcal-fetch))
#+end_src

One-time setup:
1. Create Google Cloud project → enable Calendar API → create OAuth2 credentials
2. Extract client_id and client_secret from credentials.json
3. Run: =gpg --edit-key YOUR_KEY_ID= to ensure you have a GPG key
4. Write: =M-x auth-source-pass-enable= then edit =~/.authinfo=:
   =machine org-gcal login client-id password YOUR_ID=
   =machine org-gcal login client-secret password YOUR_SECRET=
5. Encrypt: =gpg --symmetric ~/.authinfo= → saves as =~/.authinfo.gpg=
6. First sync: =M-x org-gcal-fetch=

* Citations (citar)

Book and paper metadata lives in =~/org/library.bib=.
=citar= provides fuzzy search over title/author for inserting citations.
=org-roam-bibtex= links .bib entries to org-roam notes.

BibTeX book entry format:
#+begin_example
@book{feynman1964lectures,
  author    = {Feynman, Richard P. and Leighton, Robert B.},
  title     = {The Feynman Lectures on Physics},
  year      = {1964},
  publisher = {Addison-Wesley},
  volume    = {1},
  keywords  = {physics, classical},
}
#+end_example

A single library.bib scales fine to 500+ books — citar searches by author/title
in milliseconds. For the visual book grid, see =~/org/personal/library.org=.

#+begin_src emacs-lisp
(use-package citar
  :custom
  (citar-bibliography '("~/org/library.bib"))
  (citar-notes-paths  '("~/org/personal/sources/books/"
                        "~/org/personal/sources/papers/"))
  :bind ([remap org-cite-insert] . citar-insert-citation))

(use-package org-roam-bibtex
  :after (org-roam citar)
  :config (org-roam-bibtex-mode))
#+end_src

* Music Player (EMMS)

In-Emacs music player backed by mpv (already installed).
Library: Artist / Album / FLAC tracks at
=/home/thijmen/Documents/BACKUP/Music Library/=.

SPC m SPC = pause/play. SPC t s = toggle sidebar.

#+begin_src emacs-lisp
(use-package emms
  :config
  (require 'emms-setup)
  (require 'emms-browser)
  (emms-all)

  ;; mpv as the sole player backend (already installed system-wide)
  (setq emms-player-list '(emms-player-mpv))
  (setq emms-player-mpv-parameters
        '("--no-terminal" "--audio-display=no"))

  ;; Library root
  (setq emms-source-file-default-directory
        "/home/thijmen/Documents/BACKUP/Music Library/")

  ;; Show "Artist — Title" in EMMS and mode-line
  (setq emms-track-description-function
        (lambda (track)
          (let ((artist (emms-track-get track 'info-artist "Unknown"))
                (title  (emms-track-get track 'info-title
                          (file-name-sans-extension
                           (file-name-nondirectory
                            (emms-track-get track 'name ""))))))
            (format "%s — %s" artist title))))

  ;; Show current track in mode-line
  (emms-mode-line-mode 1)
  (emms-playing-time-mode 1))

;; Sidebar toggle: opens EMMS browser in a 32-column right window
(defun emms-toggle-sidebar ()
  "Toggle the EMMS browser in a dedicated right-side window."
  (interactive)
  (if-let ((win (get-buffer-window emms-browser-buffer-name)))
      (delete-window win)
    (let ((cur (selected-window)))
      (select-window (split-window-right -32))
      (emms-browser)
      (set-window-dedicated-p (selected-window) t)
      (select-window cur))))
#+end_src

To scan the library on first use:
1. =M-x emms-add-directory-tree= → enter =/home/thijmen/Documents/BACKUP/Music Library/=
2. EMMS caches the library; subsequent startups load instantly via =emms-cache=.
   Add =(emms-cache-enable)= below if you want persistent cache across restarts.

* Org Directory Initialization

Run =M-x org-setup= on first Emacs startup to create the =~/org/= directory structure.

#+begin_src emacs-lisp
(defun org-setup ()
  "Create the ~/org/ directory structure and initial files if they don't exist."
  (interactive)
  (let ((dirs '("~/org/personal/concepts"
                "~/org/personal/knowledge/physics"
                "~/org/personal/knowledge/philosophy"
                "~/org/personal/knowledge/mathematics"
                "~/org/personal/knowledge/languages"
                "~/org/personal/essays"
                "~/org/personal/sources/books"
                "~/org/personal/sources/papers"
                "~/org/personal/sources/youtube"
                "~/org/personal/people/known"
                "~/org/personal/people/famous"
                "~/org/personal/misc"
                "~/org/uni/classes"
                "~/org/uni/lectures"
                "~/org/uni/assignments"
                "~/org/uni/summary"
                "~/org/uni/thesis"
                "~/org/daily")))
    (dolist (dir dirs) (make-directory dir t)))

  ;; Create stub files
  (let ((stubs
         '(("~/org/inbox.org"
            "#+title: Inbox\n\nProcess regularly with SPC n c → i.\n")
           ("~/org/agenda.org"
            "#+title: Agenda\n\n* Tasks\n")
           ("~/org/library.bib"
            "% BibTeX library — add books and papers here.\n% citar searches this file.\n")
           ("~/org/personal/dashboard.org"
            "#+title: Personal\n#+startup: content\n\n* Inbox\n#+BEGIN: org-ql-table :query (todo \"TODO\") :from \"~/org/inbox.org\"\n#+END:\n\n* Recent notes\n#+BEGIN: org-ql-table :query (ts :from -7) :columns (file ts) :from \"~/org/personal/\"\n#+END:\n")
           ("~/org/uni/dashboard.org"
            "#+title: Uni\n#+startup: content\n\n* Upcoming deadlines\n#+BEGIN: org-ql-table :query (and (tags \"assignment\") (not (property \"grade\")) (deadline :from today :to +30)) :columns (file (property \"deadline\") (property \"class\")) :from \"~/org/uni/\"\n#+END:\n\n* Classes\n#+BEGIN: org-ql-table :query (tags \"class\") :columns (file (property \"year\") (property \"Q\")) :from \"~/org/uni/classes/\"\n#+END:\n"))))
    (pcase-dolist (`(,path ,content) stubs)
      (let ((expanded (expand-file-name path)))
        (unless (file-exists-p expanded)
          (with-temp-file expanded (insert content))))))

  (message "org/ structure created. Run M-x org-roam-db-sync to build the database."))
#+end_src

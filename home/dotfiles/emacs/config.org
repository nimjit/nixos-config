** Config hot-reload

Save config.org → retangle to config.el → hot-reload in the running Emacs.
Most settings and keybindings are idempotent; use-package forms re-run safely
(all packages come from Nix, so =:ensure nil= means no download attempts).
Use =SPC q R= for changes that need a fresh process (new packages, etc.).

#+begin_src emacs-lisp
(defun my/config-retangle ()
  "Retangle config.org → config.el and hot-reload it (called from after-save-hook)."
  (when (equal (buffer-file-name)
               (expand-file-name "config.org" user-emacs-directory))
    (let ((config-el (expand-file-name "config.el" user-emacs-directory)))
      (org-babel-tangle)
      (condition-case err
          (load-file config-el)
        (error (message "config reload error: %s" err)))
      (message "config.org retangled and reloaded"))))

(add-hook 'after-save-hook #'my/config-retangle)

(defun my/reload-config ()
  "Manually retangle config.org and hot-reload config.el (=SPC q r=)."
  (interactive)
  (let* ((dir user-emacs-directory)
         (org (expand-file-name "config.org" dir))
         (el  (expand-file-name "config.el"  dir)))
    (with-current-buffer (find-file-noselect org)
      (org-babel-tangle))
    (condition-case err
        (load-file el)
      (error (message "config reload error: %s" err)))
    (message "Config reloaded.")))

(defun my/restart-daemon ()
  "Save all buffers and restart the Emacs daemon via systemctl (=SPC q R=).
The current client frame will die; reconnect with: emacsclient -c"
  (interactive)
  (save-some-buffers t)
  (start-process "restart-emacs-daemon" nil
                 "systemctl" "--user" "restart" "emacs")
  (message "Daemon restarting — reconnect with: emacsclient -c"))
#+end_src

        (setq my/markdown-reading-active t)
        ;; Force code blocks to pull syntax highlighting directly from your active theme
        (setq markdown-fontify-code-blocks-natively t) 
        ;; Hide the markdown symbols directly via the backend variable
        (setq markdown-hide-markup t)
        (font-lock-flush)
        (font-lock-ensure)
        ;; Render embedded graphics safely
        (when (fboundp 'markdown-display-inline-images)
          (ignore-errors (markdown-display-inline-images)))

        (evil-motion-state)                 ; Safe scroll state (j/k, d/u work perfectly)

** Typst Math Preview

Custom implementation rendering =$...$= Typst fragments as inline SVG overlays.
Cursor leaves fragment → renders via =typst compile --format svg=.
Cursor enters fragment → shows source for editing.

This mirrors typst-mate behavior in Obsidian, using the Typst CLI already installed.

#+begin_src emacs-lisp
(defgroup org-typst-preview nil
  "Preview Typst math $...$ fragments inline in org and markdown buffers."
  :group 'org)

(defcustom org-typst-preview-scale 1.4
  "SVG scale factor for rendered math. 1.4 ≈ 13pt typst at 14pt body font."
  :type 'number :group 'org-typst-preview)

;; Ukiyo base05 (main text color) — update if you change themes
(defcustom org-typst-preview-text-color "#ccc2b7"
  "Typst text color for rendered fragments; should match your theme."
  :type 'string :group 'org-typst-preview)

(defvar-local org-typst-preview--last-point nil)
(defvar-local org-typst-preview--idle-timer nil
  "Pending re-render timer, cancelled if cursor re-enters fragment before firing.")

(defun org-typst-preview--find-fragment (pos)
  "Return (beg . end) of the $...$ fragment containing POS, or nil.
Only matches single-line fragments without nested dollar signs."
  (save-excursion
    (let ((line-beg (line-beginning-position))
          (line-end (line-end-position))
          result)
      (goto-char line-beg)
      (while (and (not result)
                  (re-search-forward "\\$\\([^$\n]+\\)\\$" line-end t))
        (when (and (>= pos (match-beginning 0))
                   (<= pos (match-end 0)))
          (setq result (cons (match-beginning 0) (match-end 0)))))
      result)))

(defun org-typst-preview--render (beg end)
  "Compile the Typst fragment at BEG..END to SVG and display as an overlay."
  (dolist (ov (overlays-in beg end))
    (when (overlay-get ov 'org-typst-preview) (delete-overlay ov)))
  (let* ((fragment (buffer-substring-no-properties beg end))
         (tmp-dir  (make-temp-file "org-typst-" t))
         (typ-file (expand-file-name "math.typ" tmp-dir))
         (svg-file (expand-file-name "math.svg" tmp-dir)))
    (with-temp-file typ-file
      (insert "#set page(width: auto, height: auto, margin: (x: 2pt, y: 3pt), fill: none)\n")
      (insert (format "#set text(size: 13pt, fill: rgb(\"%s\"))\n"
                      org-typst-preview-text-color))
      (insert fragment "\n"))
    (when (zerop (call-process "typst" nil nil nil
                               "compile" "--format" "svg" typ-file svg-file))
      (let* ((paged (concat (file-name-sans-extension svg-file) "1.svg"))
             (actual-svg (cond ((file-exists-p svg-file) svg-file)
                               ((file-exists-p paged)    paged))))
        (when actual-svg
          (let ((ov (make-overlay beg end)))
            (overlay-put ov 'display
                         (create-image actual-svg 'svg nil
                                       :scale org-typst-preview-scale))
            (overlay-put ov 'org-typst-preview t)
            (overlay-put ov 'evaporate t)
            (run-at-time 30 nil (lambda () (ignore-errors (delete-directory tmp-dir t))))))))))

(defun org-typst-preview-buffer ()
  "Render all $...$ fragments in the current buffer."
  (interactive)
  (dolist (ov (overlays-in (point-min) (point-max)))
    (when (overlay-get ov 'org-typst-preview) (delete-overlay ov)))
  (let ((orig (point)))
    (save-excursion
      (goto-char (point-min))
      (while (re-search-forward "\\$\\([^$\n]+\\)\\$" nil t)
        (let ((beg (match-beginning 0)) (end (match-end 0)))
          (unless (and (>= orig beg) (<= orig end))
            (org-typst-preview--render beg end)))))))

(defun org-typst-preview--post-command ()
  "Auto-toggle preview when cursor moves in or out of a math fragment."
  (when (and (memq major-mode '(org-mode markdown-mode)) (not (minibufferp)))
    (let ((prev org-typst-preview--last-point)
          (curr (point)))
      (setq org-typst-preview--last-point curr)
      (when prev
        (let ((prev-frag (org-typst-preview--find-fragment prev))
              (curr-frag (org-typst-preview--find-fragment curr)))
          ;; Moved INTO a fragment → remove overlay immediately to show source
          (when (and curr-frag (not (equal curr-frag prev-frag)))
            (when org-typst-preview--idle-timer
              (cancel-timer org-typst-preview--idle-timer)
              (setq org-typst-preview--idle-timer nil))
            (dolist (ov (overlays-in (car curr-frag) (cdr curr-frag)))
              (when (overlay-get ov 'org-typst-preview) (delete-overlay ov))))
          ;; Moved OUT of a fragment → re-render after 0.3s idle (debounced)
          (when (and prev-frag (not (equal prev-frag curr-frag)))
            (when org-typst-preview--idle-timer
              (cancel-timer org-typst-preview--idle-timer))
            (let ((buf (current-buffer))
                  (beg (car prev-frag))
                  (end (cdr prev-frag)))
              (setq org-typst-preview--idle-timer
                    (run-with-idle-timer 0.3 nil
                      (lambda ()
                        (when (buffer-live-p buf)
                          (with-current-buffer buf
                            (setq org-typst-preview--idle-timer nil)
                            (org-typst-preview--render beg end)))))))))))))

(define-minor-mode org-typst-preview-mode
  "Auto-preview Typst math $...$ fragments in org and markdown buffers."
  :lighter " TypPrev"
  (if org-typst-preview-mode
      (progn
        (add-hook 'post-command-hook #'org-typst-preview--post-command nil t)
        (org-typst-preview-buffer))
    (when org-typst-preview--idle-timer
      (cancel-timer org-typst-preview--idle-timer)
      (setq org-typst-preview--idle-timer nil))
    (remove-hook 'post-command-hook #'org-typst-preview--post-command t)
    (dolist (ov (overlays-in (point-min) (point-max)))
      (when (overlay-get ov 'org-typst-preview) (delete-overlay ov)))))

;; Auto-start in org and markdown buffers
(add-hook 'org-mode-hook      #'org-typst-preview-mode)
(add-hook 'markdown-mode-hook #'org-typst-preview-mode)
#+end_src

** YouTube view

Two-panel YouTube feed viewer. =Y= on dashboard opens it. Category filter via =c=.
Thumbnail images displayed inline in the preview panel (GUI only).
Shares thumbnail cache with terminal =yt-feed= (~/.cache/yt-feed/).

After adding category tags to elfeed.org: run =M-x my/yt--retag-all= once to apply
the new tags to entries already in the DB.

#+begin_src emacs-lisp
(defvar my/yt--category nil "Current category tag filter, or nil for All.")

(defconst my/yt-categories
  '(("All"           . nil)
    ("Physics"        . phys)
    ("AutoDidact"     . autodidact)
    ("Politics"       . politics)
    ("Philosophy"     . phil)
    ("Minecraft"      . minecraft)
    ("Movies & Film"  . film)
    ("Vim & Emacs"    . tech)
    ("Fashion"        . fashion)
    ("Bodybuilding"   . lifting)
    ("Old Reliable"   . oldrel)))

(defun my/yt--id (url)
  "Extract YouTube video ID from URL."
  (when (and url (string-match "\\(?:[?&]v=\\|youtu\\.be/\\)\\([A-Za-z0-9_-]\\{11\\}\\)" url))
    (match-string 1 url)))

(defun my/yt--thumb-path (id)
  (expand-file-name (concat id ".jpg") (expand-file-name "~/.cache/yt-feed/")))

(defun my/yt--thumb-url (id)
  (format "https://i.ytimg.com/vi/%s/mqdefault.jpg" id))

(defun my/yt--channel (entry)
  (let ((feed (elfeed-entry-feed entry)))
    (if feed (elfeed-feed-title feed) "?")))

(defun my/yt--entries ()
  "Return YouTube entries newest first, filtered by my/yt--category."
  (let (es)
    (with-elfeed-db-visit (entry _feed)
      (let ((tags (elfeed-entry-tags entry))
            (link (or (elfeed-entry-link entry) "")))
        (when (and (or (memq 'youtube tags)
                       (string-match-p "youtube\\.com\\|youtu\\.be" link))
                   (or (null my/yt--category)
                       (memq my/yt--category tags)))
          (push entry es))))
    (sort es (lambda (a b) (> (elfeed-entry-date a) (elfeed-entry-date b))))))

(defun my/yt--retag-all ()
  "Apply feed tags from elfeed-org to all existing DB entries.
Run once after adding category tags to elfeed.org."
  (interactive)
  (elfeed-org)
  (with-elfeed-db-visit (entry _feed)
    (let* ((feed (elfeed-entry-feed entry))
           (feed-tags (when feed (elfeed-meta feed :tags))))
      (dolist (tag feed-tags)
        (elfeed-tag entry tag))))
  (elfeed-db-save)
  (message "Done retagging all elfeed entries."))

;; ── List buffer ────────────────────────────────────────────────────────────

(defvar my/yt--list-buf    "*YouTube List*")
(defvar my/yt--preview-buf "*YouTube Preview*")
(defvar my/yt--window-config nil "Window config saved on open, restored on quit.")
(defvar my/yt--all-entries nil)

(defun my/yt--draw-list ()
  "Erase and redraw the list buffer. Returns the buffer."
  (setq my/yt--all-entries (my/yt--entries))
  (let* ((buf (get-buffer-create my/yt--list-buf))
         (cat-name (or (car (rassq my/yt--category my/yt-categories)) "All"))
         (n (length my/yt--all-entries)))
    (with-current-buffer buf
      (let ((inhibit-read-only t))
        (erase-buffer)
        (insert (propertize (format " YouTube  [%s]  (%d)\n" cat-name n) 'face 'bold))
        (insert (propertize (make-string 70 ?─) 'face 'shadow) "\n")
        (dolist (entry my/yt--all-entries)
          (let* ((title   (or (elfeed-entry-title entry) ""))
                 (channel (my/yt--channel entry))
                 (date    (format-time-string "%m-%d"
                           (seconds-to-time (elfeed-entry-date entry))))
                 (unread  (memq 'unread (elfeed-entry-tags entry)))
                 (line    (format " %s  %-20s  %s\n"
                                  date
                                  (truncate-string-to-width channel 20)
                                  (truncate-string-to-width title 55))))
            (insert (propertize line
                                'elfeed-entry entry
                                'face (if unread 'default '(:foreground "#777"))))))
        (read-only-mode 1))
      (goto-char (point-min))
      (forward-line 2))
    buf))

(defun my/yt--entry-at-point ()
  (get-text-property (line-beginning-position) 'elfeed-entry))

(defun my/yt--move (n)
  "Move N lines; skip past header lines; update preview."
  (forward-line n)
  (when (< (line-number-at-pos) 3)
    (goto-char (point-min))
    (forward-line 2))
  (my/yt--update-preview))

(defun my/yt--next () (interactive) (my/yt--move 1))
(defun my/yt--prev () (interactive) (my/yt--move -1))

(defun my/yt--play ()

  (interactive)

              (url (elfeed-entry-link e)))
    (elfeed-untag e 'unread)
    (elfeed-db-save)
    (start-process "mpv-yt" nil "mpv"
                   "--gpu-api=opengl"
                   "--title=emacs-youtube"
                   url)))

(defun my/yt--pick-category ()
  "Choose a category with completing-read (uses vertico)."
  (interactive)
  (let* ((choice (completing-read "Category: " (mapcar #'car my/yt-categories) nil t))
         (tag (cdr (assoc choice my/yt-categories))))
    (setq my/yt--category tag)
    (my/yt--draw-list)
    (my/yt--update-preview)))

(defun my/yt--refresh ()

  (interactive)

  (elfeed-update)
  (run-with-timer 8 nil (lambda ()
                          (my/yt--draw-list)
                          (my/yt--update-preview)
                          (message "YouTube: done."))))

(defun my/yt--quit ()
  "Kill YouTube buffers and restore the previous window layout."
  (interactive)

    (when (get-buffer b) (kill-buffer b)))

      (set-window-configuration my/yt--window-config)
    (delete-other-windows))
  (setq my/yt--window-config nil))

;; ── Preview buffer ─────────────────────────────────────────────────────────

(defun my/yt--update-preview ()
  "Redraw preview for the entry currently at point in the list buffer."
  (when (buffer-live-p (get-buffer my/yt--list-buf))
    (with-current-buffer my/yt--list-buf
      (when-let ((e (my/yt--entry-at-point)))
        (my/yt--draw-preview e)))))

(defun my/yt--draw-preview (entry)
  "Populate *YouTube Preview* for ENTRY."
  (let* ((buf   (get-buffer-create my/yt--preview-buf))
         (title (or (elfeed-entry-title entry) ""))
         (ch    (my/yt--channel entry))
         (date  (format-time-string "%Y-%m-%d"
                 (seconds-to-time (elfeed-entry-date entry))))
         (url   (elfeed-entry-link entry))
         (id    (when url (my/yt--id url)))
         (path  (when id (my/yt--thumb-path id))))
    (with-current-buffer buf
      (let ((inhibit-read-only t))
        (erase-buffer)
        (cond
         ((and path (file-exists-p path))
          (insert-image (create-image path 'jpeg nil :max-width 720 :max-height 480) " ")
          (insert "\n\n"))
         (id
          (insert (propertize "[loading thumbnail…]\n\n" 'face 'shadow))
          (make-directory "~/.cache/yt-feed/" t)

                     (not (string-prefix-p "." folder))
                     (or (not (gethash folder folder-mtime))
                         (> mt (gethash folder folder-mtime))))
            (puthash folder mt folder-mtime)
            (puthash folder (cons path mt) by-folder)))))
    (seq-take
     (mapcar (lambda (f)
               (let* ((e    (gethash f by-folder))

             (sort (hash-table-keys folder-mtime)

                                    (gethash b folder-mtime)))))
     n)))

(defun my/render-md-table (lines &optional max-width)
  "Convert markdown pipe-table LINES to box-drawing unicode.
When MAX-WIDTH is set, shrink widest columns so total fits."
  (let* ((is-sep   (lambda (l) (string-match-p "^[[:space:]]*|[[:space:]]*[-:]" l)))
         (cells-of (lambda (l)
                     (mapcar #'string-trim
                             (split-string (string-trim l "|" "|") "|"))))
         widths)
    ;; Measure column widths from data rows only
    (dolist (line lines)
      (unless (funcall is-sep line)
        (when (string-match-p "^[[:space:]]*|" line)
          (let ((i 0))
            (dolist (cell (funcall cells-of line))
              (when (>= i (length widths))
                (setq widths (append widths (list 0))))
              (setf (nth i widths) (max (nth i widths) (length cell)))
              (setq i (1+ i)))))))
    ;; Shrink widest columns proportionally if table exceeds max-width
    (when (and max-width widths)
      (let* ((total (+ 1 (* 3 (length widths)) (apply #'+ widths)))
             (excess (- total max-width)))
        (when (> excess 0)
          (let ((sorted-i (sort (number-sequence 0 (1- (length widths)))
                                (lambda (a b) (> (nth a widths) (nth b widths))))))
            (dolist (i sorted-i)
              (when (> excess 0)
                (let ((shrink (min excess (max 0 (- (nth i widths) 4)))))
                  (setf (nth i widths) (- (nth i widths) shrink))
                  (setq excess (- excess shrink)))))))))
    (if (null widths)
        lines
      (mapcar
       (lambda (line)
         (cond
          ((funcall is-sep line)
           (concat "├"
                   (mapconcat (lambda (w) (make-string (+ w 2) ?─)) widths "┼")
                   "┤"))
          ((string-match-p "^[[:space:]]*|" line)
           (let* ((cs    (funcall cells-of line))
                  (parts (cl-loop for i from 0 below (length widths)
                                  for cell = (or (nth i cs) "")
                                  collect (concat " " cell
                                                  (make-string (max 0 (- (nth i widths) (length cell))) ?\s)
                                                  " "))))
             (concat "│" (string-join parts "│") "│")))
          (t line)))
       lines))))
